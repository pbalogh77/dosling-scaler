cmake_minimum_required(VERSION 3.18)
project(DoslingScaler DESCRIPTION "Dosling scaler" LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DL_ENABLE_NEON "ARM NEON comparison keys" ON)
option(DL_BUILD_TOOLS "Build offline trainer and host CLI" ON)
option(DL_SANITIZE "Memory/UB sanitizer" OFF)
option(DL_ENABLE_IPO "Interprocedural optimization in supported Release builds" ON)
add_library(dos_lut_hqx SHARED src/hq_block.c)
target_include_directories(dos_lut_hqx PRIVATE ../vendor/hqx)
add_library(dos_lut SHARED src/api.cpp src/frame.cpp src/model.cpp src/ufo.cpp src/aa_plane.cpp)
add_library(DoslingScaler::Scaler ALIAS dos_lut)
target_include_directories(dos_lut PUBLIC include PRIVATE src)
target_link_libraries(dos_lut PRIVATE dos_lut_hqx)
if(DL_ENABLE_IPO AND NOT DL_SANITIZE)
 include(CheckIPOSupported)
 check_ipo_supported(RESULT DL_IPO_SUPPORTED OUTPUT DL_IPO_DIAGNOSTIC LANGUAGES C CXX)
 if(DL_IPO_SUPPORTED)
  # Optimize the multi-file runtime. The replaceable HQx fallback stays a
  # separate, ordinarily compiled module; there is no cross-library inlining.
  set_target_properties(dos_lut PROPERTIES
   INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE
   INTERPROCEDURAL_OPTIMIZATION_RELWITHDEBINFO TRUE)
 else()
  message(STATUS "Runtime IPO unavailable; using ordinary compilation: ${DL_IPO_DIAGNOSTIC}")
 endif()
endif()
